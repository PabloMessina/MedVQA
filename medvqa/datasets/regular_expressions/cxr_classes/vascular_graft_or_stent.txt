        # aortoiliac stent, graft, endoprosthesis, endograft, or stent-graft.

        # Device-first: stent / stent-graft in a named artery (incl. LAD/RCA/LCX;
        # "origin of the" recovers ostial/innominate phrasing)
        (?<!\bno\s)(?<!\bwithout\s)(?:stent[-\s]?grafts?|stents?)\s+(?:(?:is|are|has|have|been|again|also|seen|noted)\s+)*(?:in|within|of)\s+(?:the\s+)?(?:(?:origin|ostium|proximal|distal)\s+of\s+(?:the\s+)?)?(?:(?:left|right)\s+)?(?:(?:common|internal|external)\s+)?(?:\blad\b|\brca\b|\blcx\b|coronary|carotid|subclavian|iliac|renal|femoral|pulmonary|popliteal|mesenteric|celiac|innominate|brachiocephalic|axillary)(?:\s+arter(?:y|ies))?\b|
        # TIPS (portal venous stent / stent-graft); not bare "tips"
        (?<!\bno\s)(?<!\bwithout\s)\btips\s+(?:grafts?|stents?)\b|

        (?<!\bno\s)(?<!\bwithout\s)(?<!\bs/p\s)(?:\btevar\b|\bevar\b)|

        (?<!\bno\s)(?<!\bwithout\s)(?<!\bs/p\s)(?<!\bh/o\s)(?:saphenous\s+)?bypass\s+grafts?\s+(?:(?:is|are|has|have|been|again|also|still|seen|noted|visualized|redemonstrat\w*|project\w*|present|remain\w*|stable|unchanged)\b|have\s+been\s+placed|in\s+place)
