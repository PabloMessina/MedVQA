# Pleural calcification (explicit calcific wording only).
# Prompt: pleural plaques/thickening/scarring/fibrosis/fibrothorax count ONLY when
# explicitly calcified/calcific. Do not match bare pleural plaques.
# Also: diaphragmatic/hemidiaphragm/apex calcific plaques; calcified asbestos plaques
# (prompt: calcified asbestos-related pleural plaques — "pleural" token optional).
# Tolerates common OCR/typos: plueral, califications.
# Exclude sub/infradiaphragmatic (abdominal) and bare aortic plaques.
#
# FN-fit (fn_mine_fit_shared): calcified(+laterality)+diaphragmatic|hemidiaphragmatic
# plaque; calcification of (the) hemidiaphragm(s); plaque-like calcified densit*/
# calcifications; widen calcified plaque→asbestos span; projecting along/over
# hemidiaphragm connectors. Do not chase bare pleural plaques.

(?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)(?<!\babsent\s)(?<!\bdenies\s)
(?<!\bnegative\sfor\s)
(?<!\bno\sevidence\sof\s)(?<!\bwithout\sevidence\sof\s)
(?<!\bno\ssign\sof\s)(?<!\bno\ssigns\sof\s)
\b(?:
        # 1. Pleural/plueral ... calcification(s)/califications/calcium
        pl(?:eu|ue)r[a-z]*
        (?:[\s,;()/-]+[a-z-]+){0,5}
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*|calcium)
        |
        # 2. Calcification(s)/califications/calcium ... pleural/plueral
        (?:calcif[a-z]*|calif[a-z]*|calcium)
        (?:[\s,;()/-]+[a-z-]+){0,8}
        [\s,;()/-]+pl(?:eu|ue)r[a-z]*
        (?:[\s,;()/-]+(?:in[\s,;()/-]+)?etiolog[a-z]*)?
        |
        # 3. Calcified pleural plaque/thickening/scar/fibrosis
        (?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+[a-z-]+){0,4}
        [\s,;()/-]+pl(?:eu|ue)r[a-z]*
        (?:[\s,;()/-]+[a-z-]+){0,3}
        [\s,;()/-]+(?:plaque[a-z]*|thicken[a-z]*|scar[a-z]*|fibro[a-z]*)
        |
        # 4. Calcific fibrothorax (pleural by definition; no "pleur*" token required)
        (?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+[a-z-]+){0,4}
        [\s,;()/-]+fibrothorax
        |
        fibrothorax
        (?:[\s,;()/-]+[a-z-]+){0,4}
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*|calcium)
        |
        # 5. Diaphragmatic / hemidiaphragm / diaphragm calcification (forward only)
        # Block sub-/infra-diaphragmatic.
        (?<!sub)(?<!infra)(?<!sub-)(?<!infra-)
        (?:diaphragmat\w*|hemidiaphragm\w*|diaphragm)
        (?:[\s,;()/-]+[a-z-]+){0,3}
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*|calcium)
        |
        # 5b. FN-fit: calcification of (the) hemidiaphragm(s) (narrow reverse;
        # hemidiaphragm only — avoids hepatic/capsular calc over diaphragm)
        (?:calcification|calification|calcifications|califications)
        [\s,;()/-]+of
        (?:[\s,;()/-]+the)?
        [\s,;()/-]+hemidiaphragm\w*
        |
        # 6. Lung base/diaphragm calcification (slash or space)
        lung[\s,;()/-]+base[\s,;()/-]+diaphragm\w*
        (?:[\s,;()/-]+[a-z-]+){0,2}
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*|calcium)
        |
        # 7. Calcified/calcific plaque on/along/in/projecting hemidiaphragm|diaphragm|apex
        # FN-fit: projecting|project between plaque and location
        (?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+[a-z-]+){0,3}
        [\s,;()/-]+plaque[a-z]*
        (?:[\s,;()/-]+(?:
            along|on|over|in|of|the|left|right|bilateral|both|
            appears?|to|have|has|a|an|mild|stable|unchanged|again|noted|seen|
            project(?:ing|s|ed)?
        )){0,8}
        [\s,;()/-]+(?:hemidiaphragm\w*|diaphragmat\w*|diaphragm\w*|(?:lung\s+)?apex\w*|\bapices?\b)
        |
        # 7b. FN-fit: calcified(+laterality)+diaphragmatic|hemidiaphragmatic+plaque
        (?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+(?:left|right|bilateral|both))?
        [\s,;()/-]+(?:hemi)?diaphragmatic
        [\s,;()/-]+plaque[a-z]*
        |
        # 8. Hemidiaphragm/apex ... calcified/calcific plaque (inverted order)
        (?:hemidiaphragm\w*|diaphragmat\w*|(?:lung\s+)?apex\w*)
        (?:[\s,;()/-]+[a-z-]+){0,6}
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+)?plaque[a-z]*
        |
        # 9. Calcified asbestos plaques / calcified plaques + asbestos
        # (prompt: calcified asbestos-related pleural plaques)
        (?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+[a-z-]+){0,4}
        [\s,;()/-]+asbestos
        (?:[\s,;()/-]+[a-z-]+){0,3}
        [\s,;()/-]+plaque[a-z]*
        |
        # FN-fit: widen plaque→asbestos intervening window (was {0,6})
        (?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+[a-z-]+){0,3}
        [\s,;()/-]+plaque[a-z]*
        (?:[\s,;()/-]+[a-z-]+){0,12}
        [\s,;()/-]+asbestos
        |
        asbestos
        (?:[\s,;()/-]+[a-z-]+){0,4}
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*)
        (?:[\s,;()/-]+[a-z-]+){0,3}
        [\s,;()/-]+plaque[a-z]*
        |
        # 10. Calcific densit* ... (wider gap) ... pleural scar/thicken/plaque
        # (e.g. densities projecting over apex ... apical pleural scarring)
        (?:calcif[a-z]*|calif[a-z]*)
        [\s,;()/-]+densit[a-z]*
        (?:[\s,;()/-]+[a-z-]+){0,12}
        [\s,;()/-]+pl(?:eu|ue)r[a-z]*
        (?:[\s,;()/-]+[a-z-]+){0,2}
        [\s,;()/-]+(?:plaque[a-z]*|thicken[a-z]*|scar[a-z]*|fibro[a-z]*)
        |
        # 11. FN-fit: plaque-like calcified densit*/calcifications (CXR pleural-plaque idiom)
        plaque[\s-]*like
        [\s,;()/-]+(?:calcif[a-z]*|calif[a-z]*)
)\b
