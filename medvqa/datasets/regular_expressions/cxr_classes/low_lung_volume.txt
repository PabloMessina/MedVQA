# ==============================================================================
# low_lung_volume
# Global low inspiratory lung expansion on the current CXR.
#
# Match: low/decreased/diminished/reduced/small/limited/poor lung volume(s)
# (either order), low-volume (chest) film, hypoinflation / underinflation /
# hypoexpansion / underexpansion, poorly/under/hypo-inflated lungs,
# mildly/slightly inflated (under-expansion paraphrase), shallow / poor /
# limited / inadequate / suboptimal / submaximal / incomplete inspiration
# (adj-first or subject-first, including degree/level of), decreased/
# diminished inspiration, less-than-full / not-fully / incompletely inspired,
# shallow (lung) volumes, low inspiratory volumes or effort, suboptimal lung
# inspiratory effort, volumes continue to be low / telegram volumes small,
# hypoventilatory / hypoaeration / decreased aeration (bilateral/global),
# incomplete/inadequate/poor/limited/decreased (lung|chest) expansion, not
# well/fully expanded lungs, expiratory appearance, due-to expiratory film
# or findings-sense obtained/taken in expiration. Global "volume loss in the
# lungs" (no laterality). IU XXXX inspiratory effort. Typos: lowlung volumes,
# low lung lungs.
#
# FN-fit (fn_mine_fit_shared): hunt residual paraphrases — subject-first
# inspiration is shallow/suboptimal; decreased/diminished inspiration;
# poor/suboptimal lung volumes; low-volume chest film; obtained in expiration;
# decreased lung expansion; limited chest expansion; telegram lung volumes small.
#
# Do NOT match: unilateral/focal/lobar volume loss; atelectasis alone; effusion;
# resection / small hemithorax; elevated hemidiaphragm without low-volume or
# poor-inspiration wording; bare respiratory-motion / technique limits;
# "volumes of low attenuation"; "decreased volume" without lungs; "lower lung
# volume" as anatomy (singular); "low normal" volumes; indication-only
# incentive spirometry; unilateral "poor expansion of the right lung";
# bare expiratory-view requests.
# ==============================================================================

\b(?:
        # ------------------------------------------------------------------
        # 1. Adjective-first: low|decreased|diminished|reduced|small
        #    (lung) volume(s). Block laterality before "lung" so
        #    "low right lung volume" / "decreased left lung volume" miss.
        #    Singular "volume" requires "lung" — avoid "decreased volume of
        #    pneumoperitoneum". "lower" is plural-only (anatomy vs finding).
        #    Block "low normal lung volumes".
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)(?<!not\s)
        (?:
            (?:low|decreased|diminished|reduced)\s+
            (?!(?:right|left|unilateral|normal)\s+)
            lung\s+volumes?
            |
            (?:low|decreased|diminished|reduced)\s+
            (?!(?:right|left|unilateral|normal)\s+)
            volumes
            |
            lower\s+
            (?!(?:right|left|unilateral)\s+)
            (?:lung\s+)?volumes
            |
            # singular "lower lung volume" as a finding (e.g. "since <date>"),
            # not anatomy / indication spirometry
            lower\s+lung\s+volume
            (?!\s+on\s+incentive)
            |
            small\s+lung\s+volumes?
            |
            small\s+volumes
            |
            # limited (lung) volumes (hunt residual; limited inspiration already §6)
            limited\s+
            (?!(?:right|left|unilateral)\s+)
            (?:lung\s+)?volumes?
            |
            # poor/suboptimal/inadequate lung volumes (adj gap vs low/decreased)
            (?:poor|suboptimal|inadequate)\s+
            (?!(?:right|left|unilateral)\s+)
            lung\s+volumes?
        )

        |

        # typo: LOWLUNG VOLUMES (no space — do not duplicate spaced "low lung")
        (?<!no\s)(?<!without\s)(?<!not\s)
        lowlung\s*volumes?

        |

        # low-volume (chest) film / exam / study / portable technique
        low[-\s]?volume\s+(?:chest\s+)?(?:film|exam(?:ination)?|study|radiograph(?:s)?|portable|technique)

        |

        # ------------------------------------------------------------------
        # 2. Decrease / reduction in|of (the) lung volumes
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)
        (?:(?:interval|mild|moderate|slight|significant)\s+)?
        (?:decrease|reduction)\s+(?:in|of)\s+(?:the\s+)?
        (?!(?:right|left|unilateral)\s+)
        (?:lung\s+)?volumes?

        |

        # ------------------------------------------------------------------
        # 3. Subject-first: (lung) volumes {are|is|remain|appear|…} low/…
        #    Require an explicit verb (or bare decreased/diminished/low after
        #    volumes) — avoid "volumes of low attenuation".
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)(?<!not\s)
        (?:lung\s+)?volumes?\s*,?\s*
        (?:
            (?:are|is|remain\w*|appear(?:\s+to\s+be)?|seem(?:\s+to\s+be)?|look|
               continue(?:s)?\s+to\s+be)\s+
            (?:still\s+|remain\w*\s+)?
            (?:quite|very|slightly|mildly|somewhat|quire|appreciably|
               moderately|significantly|extremely|exceedingly|relatively|
               generally|chronically|persistently|fairly|rather|markedly|
               severely)?\s*
            (?:low|lower|decreased|diminished|reduced|small)
            |
            (?:have|has)\s+(?:slightly|mildly|somewhat)?\s*
            (?:decreased|diminished|reduced)
            |
            (?:slightly|mildly|somewhat|generally|still)?\s*
            (?:decreased|diminished|reduced|lower|low|small)
            |
            which\s+remain\w*\s+(?:slightly|mildly|very|somewhat)?\s*low
        )

        |

        # IU / dictation typo: "Lung lungs remain low"
        (?<!no\s)(?<!without\s)(?<!not\s)
        lung\s+lungs?\s+remain\w*\s+(?:low|decreased|diminished|reduced)

        |

        # dictation typo: "low lung lungs" (volumes → lungs)
        (?<!no\s)(?<!without\s)(?<!not\s)
        (?:(?:mild(?:ly)?|moderate(?:ly)?|slight(?:ly)?|marked(?:ly)?|
            generalized|global)\s+)?
        low\s+lung\s+lungs\b

        |

        # ------------------------------------------------------------------
        # 4. Lungs low/small/diminished in volume; poorly/under/hypo inflated;
        #    mildly/slightly inflated (under-expansion paraphrase); not
        #    well/fully expanded (plural lungs — avoid unilateral "the lung")
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)(?<!not\s)
        (?:the\s+)?lungs?\s+
        (?:
            (?:are\s+|appear(?:\s+to\s+be)\s+|remain\w*\s+)?
            (?:still\s+)?
            (?:quite|very|slightly|mildly|somewhat)?\s*
            (?:low|small|diminished)\s+in\s+volumes?
            |
            (?:are\s+|appear(?:\s+to\s+be)\s+)?
            (?:slightly|mildly|somewhat|still|very|markedly|moderately|
               severely|chronically)?\s*
            (?:poorly|under|hypo)[-\s]?(?:inflat|expand|aerat)\w*
            |
            # "mildly/slightly/somewhat inflated" (= under-expanded), optional
            # trailing chronically (hunt: "mildly inflated chronically")
            (?:are\s+|appear(?:\s+to\s+be)\s+|remain\w*\s+)?
            (?:mildly|slightly|somewhat)\s+inflat\w*
            (?:\s+chronically)?
            |
            (?:are\s+|appear(?:\s+to\s+be)\s+)?
            (?:slightly\s+)?less\s+well\s+inflat\w*
        )

        |

        # plural lungs … not (well|fully) expanded — allow short clause gap
        # ("are clear although not well expanded"); do not consume the "not"
        (?<!no\s)(?<!without\s)
        (?:the\s+)?lungs\b
        (?:\s+(?!not\b)\w+){0,8}\s+
        not\s+(?:well|fully)\s+(?:expand|inflat)\w*

        |

        # "Lung low in volume" (singular subject without verb)
        lung\s+low\s+in\s+volumes?

        |

        # ------------------------------------------------------------------
        # 5. Stem families (bare). No spaced "hypo aerated" / "hypo ventilation"
        #    — those FNs are unilateral / lobar (prompt-forbidden).
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)(?<!not\s)
        (?:
            hypo(?:inflat|expand|inspir|aerat|ventilat)\w*
            |
            under(?:inflat|expand|aerat)\w*
            |
            underexpand\w*
        )

        |

        # ------------------------------------------------------------------
        # 6. Inspiration quality / effort / volumes / expansion paraphrases
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)(?<!not\s)
        (?:
            # shallow inspiration / breath / inflation, or shallow (lung) volumes
            (?<!normal\s)
            (?:very\s+|mildly\s+|slightly\s+|somewhat\s+|quite\s+)?
            shallow(?:er)?\s+
            (?:inspir\w*|breath(?:ing)?|lung\s+inflat\w*|(?:lung\s+)?volumes?)
            |
            # subject-first: (degree/level/depth of) inspiration is/appears
            # shallow|poor|limited|suboptimal|…
            (?:(?:the\s+)?(?:degree|level|depth)\s+of\s+)?inspirations?\s+
            (?:is|are|appears?\s+to\s+be|seems?\s+to\s+be|remains?)\s+
            (?:slightly|mildly|somewhat|still|very)?\s*
            (?:shallow|poor|limited|suboptimal|inadequate|incomplete)
            |
            # poor/limited/…/incomplete inspiration or effort; optional interposed "lung"
            (?:poor(?:er|ly)?|limited|inadequate|suboptimal|submaximal|fair|XXXX|
               incomplete)\s+
            (?:voluntary\s+)?(?:lung\s+)?(?:inspir\w*|inspired)
            |
            # less than (a) full inspiration (hunt residual)
            less\s+than\s+(?:an?\s+)?full\s+inspir\w*
            |
            # not fully / incompletely inspired (past participle residual)
            not\s+fully\s+inspir\w*
            |
            incompletely\s+inspir\w*
            |
            low\s+(?:level\s+of\s+)?inspir\w*
            |
            # decreased/diminished inspiration (not only inspiratory effort)
            (?:decreased|diminished)\s+inspirations?\b
            |
            (?:low|lower|reduced|decreased|limited)\s+
            inspirator(?:y)?\s+(?:effort|excursion|volumes?)
            |
            inspirator(?:y)?\s+(?:effort|excursion|volumes?)\s+
            (?:are|is|may\s+be|appear(?:\s+to\s+be)?)\s+
            (?:slightly|mildly)?\s*
            (?:low|lower|limited|poor|reduced|decreased)
            |
            (?:decreased|reduced|limited|shallow)\s+depth\s+of\s*inspir\w*
            |
            (?:reduced|decreased|diminished)\s+(?:inspiratory\s+)?(?:lung\s+)?expansion
            (?!\s+of\s+(?:the\s+)?(?:right|left|upper|lower|middle|post[- ]?op))
            |
            # incomplete/inadequate/poor/limited/(suboptimal)/decreased (overall)
            # (lung|chest) expansion — block "… expansion of the right/left …"
            (?:incomplete|inadequate|poor(?:er)?|limited|suboptimal|decreased|
               diminished)\s+
            (?:overall\s+)?(?:(?:lung|chest)\s+)?expansion
            (?!\s+of\s+(?:the\s+)?(?:right|left|upper|lower|middle|post[- ]?op))
            |
            # subject-first: (lung) expansion is/appears limited|poor|…
            (?:lung\s+)?expansion\s+
            (?:is|are|appears?\s+to\s+be|seems?\s+to\s+be|remains?)\s+
            (?:slightly|mildly|somewhat|still|very)?\s*
            (?:limited|poor|inadequate|incomplete|reduced|decreased|suboptimal)
            |
            # bilateral/global decrease(d)/diminished/reduced aeration
            (?:(?:interval|mild|moderate|slight|significant)\s+)?
            (?:decrease|decreased|diminished|reduction|reduced)\s+(?:in\s+)?
            aeration\s+
            (?:bilaterally|bilateral|globally|global|overall)\b
            |
            (?:bilateral(?:ly)?|global(?:ly)?|overall)\s+
            (?:decrease|decreased|diminished|reduction|reduced)\s+(?:in\s+)?
            aeration\b
            |
            expirator(?:y)?\s+appearance
            |
            # findings-sense: markings/opacity "due to expiratory film"
            # (not bare technique labels / recommended expiratory views)
            (?:due\s+to|secondary\s+to|related\s+to|from|reflect(?:s|ing)?)\s+
            (?:an?\s+)?expirator(?:y)?\s+(?:film|view|radiograph|exam(?:ination)?)
            |
            # findings-sense: film/images obtained|taken in expiration
            # (not dedicated PTX expiratory-view requests)
            (?:film|image|radiograph|exam(?:ination)?)s?\s+
            (?:(?:was|were|is|are)\s+)?
            (?:obtained|taken)\s+in\s+expiration
            (?!\s+(?:to\s+(?:evaluate|assess|exclude|rule|look)|
                    for\s+(?:(?:a|an)\s+)?(?:pneumothorax|pneumo|ptx)))
        )

        |

        # ------------------------------------------------------------------
        # 7. Global volume loss in/of the lungs (plural lungs, no laterality).
        #    Do not match "volume loss in the left lung" / hemithorax / lobe.
        # ------------------------------------------------------------------
        (?<!no\s)(?<!without\s)(?<!not\s)
        volume\s+loss\s+(?:in|of)\s+(?:the\s+)?lungs\b
)
