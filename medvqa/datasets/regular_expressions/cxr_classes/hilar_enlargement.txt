# ==============================================================================
# hilar_enlargement
# Abnormal enlargement / prominence of one or both pulmonary hila on the
# CURRENT CXR (prompt-faithful).
#
# Match: hilum/hila/hilar silhouette enlarged, prominent, bulky, plump, bulbous,
# full, widened, lobulated, convex, or increased in size; fullness / bulging of
# the hilum; hilar adenopathy or hilar mass/lesion/tumor/opacity that expands
# the silhouette; dilated/engorged hilar vessels when tied to the hilum.
#
# Do NOT match: normal hilum; isolated hilar calcification; increased hilar
# density without enlargement; retraction/elevation/displacement from volume
# loss; pulmonary vascular congestion / cephalization / edema / PAH without
# hilum-size language; mediastinal widening; non-hilar adenopathy; history /
# indication / known-prior-only; negated "no … or hilar …" lists.
#
# Fixed-width lookbehinds only (stdlib re). Eval lowercases + squashes space.
# ==============================================================================

\b(?:

        # A. Direct hilar size / contour adjectives
        # ------------------------------------------------------------------
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)(?<!\bnor\s)
        (?<!\bno\sright\s)(?<!\bno\sleft\s)(?<!\bno\sbilateral\s)(?<!\bno\sboth\s)
        (?<!\bnot\sright\s)(?<!\bnot\sleft\s)
        (?<!\bno\sdefinite\s)(?<!\bno\ssignificant\s)(?<!\bno\sobvious\s)
        (?<!\bno\sevidence\sof\s)(?<!\bwithout\sevidence\sof\s)
        (?<!\brule\sout\s)(?<!r/o\s)(?<!\bevaluate\sfor\s)
        (?<!\bhistory\sof\s)(?<!\bhx\sof\s)
        (?<!\bknown\s)
        (?:(?:right|left|bilateral|both|relative|possible|mild|slight|marked|moderate)\s+)?
        (?:bihilar|hilar)
        \s+
        (?:enlarg\w*|prominen\w*|fullness|widening|lobulat\w*|convexity)
        (?!\s*\?)

        |

        (?:bulky|prominent|abnormally\s+prominent|full|large|mildly\s+prominent|widened|lobulated|convex|chronically\s+large|plump|bulbous)

        (?:bihilar|hil(?:um|a|us))\b

        \s+(?:left\s+|right\s+|bilateral\s+)?hilar\s+
        (?:contour|silhouette|region|enlarg\w*|nodes?|adenopathy|lymphadenopathy|mass(?:es)?|opacit\w*|lesions?|fullness|structures?|shadows?)

        |

        # enlarged / enlargement of [the] [side] hilum (+ optional convex/shadow)
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        enlarg\w*\s+(?:(?:convex|lobulated|rounded)\s+)?(?:of\s+)?(?:the\s+)?(?:right\s+|left\s+|bilateral\s+|both\s+)?
        (?:bihilar|hil(?:um|a|us|ar)(?:\s+shadows?)?)\b

        |

        # hilum/hila is/are/appear/remain/look [modifiers] enlarged/prominent/larger
        (?<!\bno\s)(?<!\bnot\s)
        hil(?:um|a|us|ar)\s+

        (?:(?:slightly|mildly|markedly|substantially|moderately|relatively|persistently|increased|also|again|still|unchanged|similarly|bilateral(?:ly)?|both|somewhat|now|rather)\s+)*

        |

        hil(?:a|um)\s+have\s+enlarg\w*
        |

        |

        # [side] hilum/hilus appears/is larger (comparative size)
        (?<!\bno\s)(?<!\bnot\s)
        (?:(?:the\s+)?(?:right\s+|left\s+|bilateral\s+|both\s+)?)?
        hil(?:um|a|us)\s+(?:appear(?:s|ing)?|is|are)\s+larger\b
        |

        (?<!\bno\s)(?<!\bnot\s)
        hil(?:um|a|us)\s+demonstrates?\s+(?:[\w-]+\s+){0,6}and\s+appears?\s+
        (?:(?:slightly|mildly|markedly|moderately|somewhat)\s+)*

        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)

        increase(?:d)?(?:\s+in)?\s+size\s+of\s+(?:the\s+)?
        (?:right\s+|left\s+|bilateral\s+|both\s+)?
        (?:hil(?:um|a|us|ar)|hilar\s+(?:lymph\s+)?nodes?)\b
        |

        # prominence / fullness / bulging of [the] [side] hilum
        # optional "of" recovers "mild prominence the right hilum"
        # "at" recovers "increased prominence at the left hilum"
        # bilateral sides: "prominence of the right and left hilum"

        #                   "prominence of the right inferior hilum"
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        (?<!\bno\sevidence\sof\s)
        (?:increased\s+)?(?:prominence|fullness|bulg(?:e|ing|es)|convexity)
        \s+(?:(?:of|to|in|at|about|around)\s+)?
        (?:the\s+)?
        (?:
            (?:(?:inferior|lower|superior|upper)\s+aspect,?\s+)?
            (?:right\s+and\s+left\s+|left\s+and\s+right\s+|right\s+|left\s+|bilateral\s+|both\s+)?
            |
            (?:right\s+and\s+left\s+|left\s+and\s+right\s+|right\s+|left\s+|bilateral\s+|both\s+)?
            (?:inferior|lower|superior|upper)\s+
        )
        (?:bihilar|hil(?:um|a|us|ar)|hilar\s+(?:region|area))\b
        (?!\s+vascular)(?!\s+markings)

        |

        # prominence/fullness of X and [side] hilum (intervening list)
        # e.g. "prominence of the upper mediastinum and left hilum"
        # e.g. "fullness in the area of azygos and left hilar area"
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        (?:increased\s+)?(?:prominence|fullness)
        \s+(?:(?:of|in)\s+)?
        (?:the\s+)?
        (?:(?:area\s+of\s+)?)?
        (?:[\w-]+\s+){1,8}
        and\s+(?:the\s+)?
        (?:right\s+|left\s+|bilateral\s+|both\s+)?
        (?:hil(?:um|a|us|ar)|hilar\s+(?:region|area))\b

        |

        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        (?:right\s+|left\s+|bilateral\s+|both\s+)?
        hil(?:um|a|us)\s+(?:prominence|fullness|enlarg\w*|bulging|convexity)

        |

        # hilar contour(s) / silhouette / shadow [is/are] [more] prominent
        (?<!\bno\s)(?<!\bnot\s)
        hilar\s+(?:silhouette|contours?|structures?|shadows?)\s+(?:is\s+|are\s+|appear(?:s|ing)?\s+)?
        (?:more\s+)?(?:enlarg\w*|prominen\w*|full|widened|lobulat\w*|convex|increased)

        |

        # lobulated / convex contour of/in the hilum / hilar region
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        (?:lobulated|convex)\s+contour\s+(?:of|in|to)\s+(?:the\s+)?
        (?:right\s+|left\s+|bilateral\s+)?
        (?:hil(?:um|a|us)|hilar\s+region)\b

        |

        (?:borderline|increased|enlarged)\s+diameter\s+of\s+(?:the\s+)?hilar\s+structures
        |

        # "prominent X and right hila" / "abnormally prominent hila" (not hilar*)
        (?<!\bno\s)(?<!\bnot\s)
        (?:abnormally\s+|possible\s+|possibility\s+of\s+)?
        prominent\s+
        (?:(?:[\w-]+\s+){0,4}and\s+(?:the\s+)?)?
        (?:right\s+|left\s+|bilateral\s+|both\s+)?hila\b

        |

        prominence\s+of\s+(?:the\s+)?(?:[\w-]+\s+){0,10}as\s+well\s+as\s+(?:both\s+|the\s+)?hila\b

        |

        # prominent caliber of the hila (vascular enlargement at hilum)
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        prominent\s+caliber\s+of\s+(?:the\s+)?hila\b

        |

        # ------------------------------------------------------------------
        # B. Hilar adenopathy / lymphadenopathy (incl. station lists)
        # ------------------------------------------------------------------
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)(?<!\bnor\s)(?<!\bor\s)
        (?<!\bno\sdefinite\s)(?<!\bno\ssignificant\s)(?<!\bno\sobvious\s)
        (?<!\bno\sevidence\sof\s)(?<!\bno\sevidence\sof\ssignificant\s)
        (?<!\bwithout\sevidence\sof\s)(?<!\bno\sevidence\sof\smediastinal\s)
        (?<!\brule\sout\s)(?<!r/o\s)
        (?<!\bhistory\sof\s)(?<!\bhx\sof\s)(?<!\bknown\s)
        (?:(?:right|left|bilateral|both|mediastinal\s+and)\s+)?
        (?:bihilar|hilar)
        (?:
            (?:,\s*(?:and\s+)?|\s+and\s+|\s+or\s+)
            (?:(?:right|left|bilateral)\s+)?
            (?:paratracheal|subcarinal|mediastinal|pretracheal|suprahilar|infrahilar|\w+\s+lobe)

        \s+(?:metastatic\s+)?(?:lymph[\s-]?)?adeno(?:pathy|pathic|apathy)

        # hilar … metastases (station lists), expands the hilar silhouette

        (?<!\brule\sout\s)(?<!r/o\s)(?<!\bknown\s)
        (?:(?:right|left|bilateral|both)\s+)?hilar
        (?:
            (?:,\s*(?:and\s+)?|\s+and\s+|\s+or\s+)
            (?:(?:right|left|bilateral)\s+)?
            (?:paratracheal|subcarinal|mediastinal|pretracheal|\w+\s+lobe)
        ){0,4}
        \s+metastas\w*

        |

        # adenopathy / lymphadenopathy in/at/involving [the] [side] hilum/hilus
        # e.g. "Adenopathy at least in the left hilus is evident."
        # e.g. "Lymphadenopathy involving the right hilum"
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)(?<!\bnor\s)
        (?<!\bno\sevidence\sof\s)(?<!\brule\sout\s)(?<!r/o\s)
        (?<!\bhistory\sof\s)(?<!\bhx\sof\s)(?<!\bknown\s)
        (?:lymph[\s-]?)?adeno(?:pathy|pathic|apathy)

        \s+(?:in|at|involving)\s+(?:the\s+)?
        (?:right\s+|left\s+|bilateral\s+|both\s+)?
        hil(?:um|a|us)\b

        |

        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)

        # hilar nodes are enlarged (anatomy-first; requires size copula)
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        hilar\s+(?:lymph\s+)?nodes?\s+
        (?:are\s+|is\s+|appear(?:s|ing)?\s+)
        (?:(?:slightly|mildly|markedly|moderately|somewhat)\s+)*
        enlarg\w*
        |

        # hilar soft-tissue bulk / adenopathy proxy
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)
        (?:(?:right|left|bilateral)\s+)?hilar\s+soft\s+tissue
        (?:s\s+that\s+may\s+represent|\s+that\s+may\s+represent|\s+densit\w*|\s+fullness)

        |

        # hilar lymph nodes framed with vascular/adenopathy DDx
        # (bare "calcified hilar lymph nodes" is NOT enlargement — do not match)
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)(?<!\bor\s)
        (?<!\bcalcified\s)(?<!\bcalcific\s)
        (?<!\bhistory\sof\s)(?<!\bhx\sof\s)(?<!\bknown\s)
        (?:(?:right|left|bilateral|both)\s+)?hilar\s+lymph\s+nodes?
        \s+or\s+enlarged\s+(?:central\s+)?pulmonary\s+arter\w*

        |

        # ------------------------------------------------------------------
        # C. Hilar mass / lesion / tumor / opacity expanding the silhouette
        # ------------------------------------------------------------------
        (?<!\bno\s)(?<!\bnot\s)(?<!\bwithout\s)(?<!\bnor\s)(?<!\bor\s)
        (?<!\bno\sdefinite\s)(?<!\bno\ssignificant\s)
        (?<!\bno\sevidence\sof\s)(?<!\bno\sevidence\sof\smediastinal\s)
        (?<!\brule\sout\s)(?<!r/o\s)
        (?<!\bhistory\sof\s)(?<!\bhx\sof\s)
        (?<!\bknown\s)(?<!\bpatient's\sknown\s)(?<!\bpatient\sknown\s)
        (?<!\breduction\sin\ssize\sof\s)(?<!\breduction\sof\s)
        (?<!\ssize\sof\sleft\s)(?<!\ssize\sof\sright\s)(?<!\ssize\sof\sbilateral\s)
        (?<!\ssize\sof\sthe\sleft\s)(?<!\ssize\sof\sthe\sright\s)
        (?:(?:right|left|bilateral)\s+)?hilar\s+
        (?:(?:[\w-]+\s+){0,3})?
