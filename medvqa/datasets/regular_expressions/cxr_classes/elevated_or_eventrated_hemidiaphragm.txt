        # FN-fit (fn_mine_fit_shared): eventur* typo; upward/ascending; high (not only

            # "increased eventuration of the right hemidiaphragm"

            (?:[^\w.!?;:]+\w+){0,6}[^\w.!?;:]+
            (?:the\s+)?(?:right|left|bilateral)?\s*
            (?:(?:(?:medial|lateral|central|anterior|posterior)\s+(?:and\s+)?){0,3})?

            # "Moderate elevation of the left-sided diaphragm",
            # "Raised contour of the right diaphragm"
            # Require laterality so bilateral "elevated diaphragms" from body habitus /

            (?:elevat\w*|event(?:rat|ur)\w*|high[-\s]?riding|rais(?:ed|ing))
            (?:[^\w.!?;:]+\w+){0,6}[^\w.!?;:]+
            (?:the\s+)?(?:right|left)(?:-sided)?\s+
            (?:(?:dome|aspect|contour)\s+of\s+(?:the\s+)?)?
            (?:(?:(?:medial|lateral|central|anterior|posterior)\s+(?:and\s+)?){0,3})?
            (?:hemi[-\s]?)?diaphragm\w*
            (?!\s+has\s+resolved)
            (?!\s+as\s+seen\s+on\s+prior)

        # hemi[ -]?diaphragm … elevat / eventrat|eventur / high-riding / raised / peaking / tenting
        # covers: "hemidiaphragm appears to be elevated",
        # "right hemidiaphragm suggest eventration",
        # "The right hemidiaphragm is raised."
        (?<!no\s)(?<!without\s)
        (?:(?:right|left|bilateral)\s+)?hemi[-\s]?diaphragm\w*
        (?:[^\w.!?;:]+(?!(?:without|no|not|denies|absent|resolved)\b)\w+){0,6}[^\w.!?;:]+
        (?<!decreased\s)(?<!decrease\s)(?<!less\s)(?<!reduced\s)

        # covers: "tenting of the lateral right hemidiaphragm",
        # "tenting of the medial aspect of the right diaphragm",
        # "tenting of the diaphragm",
        # "tenting along the left hemidiaphragm",
        # "Tenting with the right hemidiaphragm.",
        # FN-fit: "Mild tenting of the left lateral hemidiaphragm",
        # "Peaking of the mid portion of the hemidiaphragm",
        # "Peaking of the right mid diaphragm"
        (?<!no\s)(?<!without\s)

        (?:(?:right|left)\s+)?
        (?:(?:medial|lateral|central|anterior|posterior|mid(?:dle)?)\s+)?
        (?:aspect\s+of\s+(?:the\s+)?)?
        (?:(?:right|left)\s+)?
        (?:hemi[-\s]?)?diaphragm\w*

        (?<!no\s)(?<!without\s)(?<!not\s)
