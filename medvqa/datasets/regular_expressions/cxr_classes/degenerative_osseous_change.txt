(?<!\bno\s)(?<!\bwithout\s)(?<!\bdenies\s)(?<!\babsent\s)(?<!non-)
(?<!\bno\ssignificant\s)(?<!\bwithout\ssignificant\s)
(?<!\bno\sdefinite\s)(?<!\bno\sobvious\s)(?<!\bno\sevidence\sof\s)
(?<!history\sof\s)(?<!hx\sof\s)(?<!h\/o\s)
\b(?:
        # Degenerative + optional short anatomic/severity fillers + finding noun.
        # Include spurr(ing)/spurs/spine/wedging/arthropathy (prompt-faithful).
        degenerative\s+(?:\w+[-\s]*){0,5}(?:changes?|disease|narrowing|arthrosis|arthritis|arthropath\w*|spurr(?:ing)?|spurs?|spine|wedging)|

        # Degenerative AC / shoulder / glenohumeral joints (no changes/disease noun)
        degenerative\s+(?:(?:bilateral|both|left|right|mild|moderate|severe)\s+)?
            (?:ac|acromioclavicular|glenohumeral|shoulder)\s+joints?|

        # Word-order variant: disc/joint/endplate degeneration; degeneration of the spine/shoulder
        (?:disc|disk|joint|endplate)(?:\s+and\s+(?:disc|disk|endplate))?\s+degeneration|
        degenerat(?:ion|ed|ive)\s+of(?:\s+the)?(?:\s+(?:bilateral|both|left|right|mild|moderate|severe|thoracic|lumbar|cervical|thoracolumbar))*\s+
            (?:spine|vertebra\w*|discs?|disks?|joints?|shoulders?|glenohumeral|acromioclavicular|ac\s+joints?)|
        degenerated\s+(?:(?:thoracic|lumbar|cervical|thoracolumbar)\s+)?(?:spine|vertebra\w*)|

        # Spondylosis / spondylotic / common "spondylitic" misspelling
        spondylos(?:is|es)?|
        spondylot(?:ic|ic)(?:\s+changes?)?|
        spondylitic(?:\s+changes?)?|

        # OA / osteoarthrosis with joint or spine context (avoid indication-only "osteoarthritis")
        (?:
            (?:glenohumeral|acromioclavicular|shoulder|spinal|spine|vertebral|facet|
               cervical|thoracic|lumbar|joint)\s+osteoarthr\w*
            |
            osteoarthr\w*\s+of(?:\s+the)?(?:\s+(?:bilateral|right|left|bilat))?\s+
                (?:spine|shoulder|glenohumeral|acromioclavicular|ac\s+joints?|cervical|thoracic|
                   lumbar|joints?|vertebra\w*)
            |
            osteoarthr\w*\s+(?:changes?|disease)
            |
            # IU de-id: "left XXXX XXXX osteoarthritis"
            (?:XXXX\s+){1,3}osteoarthr\w*
        )|

        osteophyt\w*|
        # DISH / AS bridging (prompt includes DISH; syndesmophyte is the spur)
        syndesmophyt\w*|
        arthritic\s+(?:changes?|disease)|

        # Arthrosis (OA synonym) at CXR-FOV joints; word orders the OA arm missed
        (?:acromioclavicular|ac|glenohumeral|shoulder|facet)\s+(?:joint\s+)?arthrosis|
        arthrosis\s+of(?:\s+the)?(?:\s+(?:bilateral|right|left|mild|moderate|severe))?\s+
            (?:ac|acromioclavicular|glenohumeral|shoulder|spine|facet)\s*joints?|

        # Disc/joint-space narrowing only with degenerative framing (prompt forbids bare narrowing)
        degenerative\s+(?:disc|disk|joint)[-\s]?space\s+narrowing|
        (?:disc|disk|joint)[-\s]?space\s+narrowing(?:\s+\w+){0,8}\s+degenerat\w*|
        degenerat\w+(?:\s+\w+){0,8}\s+(?:disc|disk|joint)[-\s]?space\s+narrowing|
        # Degenerative disc height loss ≡ degenerative disc-space narrowing
        degenerative\s+(?:disc|disk)\s+height\s+loss|

        # Degenerative endplate (not bare scalloping/sclerosis)
        degenerative\s+endplate(?:\s+\w+){0,3}|

        # DISH
        \bdish\b|
        diffuse\s+idiopathic\s+skeletal\s+hyperostosis|

        # Common acronyms
        \bdjd\b|
        \bddd\b|

        # Joint arthropathy on CXR FOV (AC / glenohumeral / shoulder / facet)
        (?:acromioclavicular|ac(?:\s+joints?)?|glenohumeral|shoulder|facet)\s+arthropath\w*|
        arthropath\w*\s+(?:in|of)\s+(?:both\s+|bilateral\s+|the\s+)?(?:shoulders?|glenohumeral|acromioclavicular|ac\s+joints?|facet)|

        # Osteophyte / spurring language (spine + CXR-FOV joints). Avoid bare
        # calcaneal/patellar/rib-only spurs and bare endplate sclerosis.
        # Word-order: vertebral/spine/body/endplate/bone/thoracic spurring.
        (?:(?:(?:mid)?thoracic|lumbar|cervical|thoracolumbar)\s+)?vertebral(?:\s+body)?\s+spurr(?:ing|s)|
        (?:thoracic|lumbar|cervical|thoracolumbar)\s+spine\s+spurr(?:ing|s)|
        # Thoracic/lumbar/cervical spurring without requiring spine/vertebral token
        (?:(?:mid)?thoracic|lumbar|cervical|thoracolumbar)\s+spurr(?:ing|s)|
        (?:(?:focal|mild|minimal|scattered)\s+)?endplate\s+spurr(?:ing|s)|
        # Spurring (+ short fillers) of/in/along/over/at/through/throughout/involving spine
        (?:(?:mild|minimal|scattered|anterior|posterior|bone|hypertrophic|arthritic|osseous|marginal)\s+)*
            spurr(?:ing|s)
            (?:\s+(?:noted|is|are|seen|present|anteriorly|posteriorly|inferiorly|superiorly|bilaterally))*
            \s+(?:of|along|in|over|at|through|throughout|involving)\s+(?:(?:the|a)\s+)?
            (?:(?:visualized|visualised)\s+)?
            (?:
                (?:(?:mid|lower|upper|thoracic|lumbar|cervical|thoracolumbar)\s+(?:and\s+(?:lower|mid|upper)\s+)?)?
                (?:thoracic\s+|lumbar\s+|cervical\s+)?(?:spine|vertebra\w*|thorax)
                |
                # mid/lower/upper T-spine / L-spine abbreviations
                (?:(?:mid|lower|upper)\s+)?[TLC][-\s]?spine
                |
                # Level-only vertebral spurring (T12-L1, T7, etc.)
                [TLC]\d+(?:\s*[-–]\s*(?:[TLC])?\d+)?
            )|
        # Arthritic / AC / shoulder / glenohumeral / clavicle / subacromial spurring (FOV joints)
        arthritic\s+spurr(?:ing|s)|
        (?:(?:mild|minimal|scattered|anterior|posterior|inferior|superior|bilateral|bone|hypertrophic|arthritic|osseous|marginal)\s+)*
            spurr(?:ing|s)
            (?:\s+(?:noted|is|are|seen|present|anteriorly|posteriorly|inferiorly|superiorly|bilaterally))*
            \s+(?:of|along|in|over|at|through|throughout|involving)\s+(?:(?:the|a)\s+)?
            (?:(?:\w+\s+){0,8})?
            (?:(?:bilateral|both|left|right|distal|proximal)\s+)*
            (?:ac\s*joints?|acromioclavicular(?:\s+joints?)?|glenohumeral(?:\s+joints?)?|shoulders?|clavicles?|coracoids?|subacromial)|
        # Noun-final FOV joint / subacromial / clavicle spurring
        (?:(?:left|right|bilateral|both)\s+)?
            (?:ac\s*joints?|acromioclavicular(?:\s+joints?)?|glenohumeral(?:\s+joints?)?|shoulders?|clavicles?|coracoids?|subacromial)\s+spurr(?:ing|s)|
        hypertrophic\s+spurr(?:ing|s)|
        hypertrophic\s+changes?\s+(?:in|of|along|at|over)\s+(?:the\s+)?
            (?:(?:left|right|bilateral|both)\s+)?
            (?:spine|vertebra\w*|thoracic|ac\s*joints?|acromioclavicular(?:\s+joints?)?|glenohumeral(?:\s+joints?)?|shoulders?|clavicles?)|

        # Elliptical: glenohumeral/AC/shoulder joints appear/are degenerative
        (?:glenohumeral|acromioclavicular|ac|shoulder)\s+joints?\s+
            (?:appear|are|look(?:s|ed)?)\s+degenerat\w*|

        # Senescent skeletal change (prompt-equivalent age-related DJD wording)
        senescent\s+(?:osseous|skeletal|bony|bone|spinal|spine|vertebral)\s+changes?|
        skeletal\s+structures?\s+(?:show|demonstrate|exhibit)\s+senescent|

        # Chronic/advanced bony change localized to a CXR-FOV joint (not postsurgical alone)
        (?:chronic|advanced)\s+(?:bony|osseous)\s+changes?\s+(?:at|of|in)\s+(?:the\s+)?
            (?:(?:left|right|bilateral)\s+)?(?:shoulder|acromioclavicular|glenohumeral)|

        # AC osteophyte-like proliferation
        proliferation\s+at\s+(?:(?:the\s+)?(?:bilateral|both)\s+)?acromioclavicular
    )
