    # 2. Descriptive Air & Fluid Co-occurrence (~150 character window)
    # Catches non-standard phrasing: "air-fluid level", "air and fluid collection", 
    # "more fluid than air", or "air inclusion... abundant fluid".
    (?:
        \b(?:air|gas)\b[\s\S]{0,150}\b(?:fluid|liquid)\b
      | \b(?:fluid|liquid)\b[\s\S]{0,150}\b(?:air|gas)\b
    )

    |
