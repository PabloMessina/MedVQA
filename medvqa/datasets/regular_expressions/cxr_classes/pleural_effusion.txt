# pleural_effusion — residual mid-node polish (stamp 20260811_030248)
# Prompt: pleural effusion/fluid/hydrothorax/hemothorax/empyema/chylothorax +
# fluid of hydro; meniscus/CPA only when attributed to fluid.
# Do NOT infer from thickening/plaques/scarring, elevated hemidiaphragm,
# consolidation, atelectasis, edema, pericardial effusion, or ascites alone.
# Pneumothorax is not an effusion. History/indication-only does not count.
#
# Children (readonly): hydropneumothorax, loculated_or_fissural_pleural_effusion.
# Residual parent: questionable/suspect hedges, compatible-with/change-in
# effusion, slash-list lookbehind bug, presumed (not "not appreciable").
# FN-fit (fn_mine_fit_shared): fluid collection in/within pleural space or
# hemithorax; parapneumonic/adjacent/exudative/transudative/hemorrhagic as
# bare-effusion cues; malignant only on compatible/consistent-with (bare
# malignant FP'd post-thora "control this malignant effusion"); scan-through
# clause-initial suspect + concerning-for pneumonia so hedged pleural fluid
# / parapneumonic / adjacent effusion hit.
# FN-fit wave29 (fn_mine_fit_shared): no (interval) increase|decrease in PE;
# pleural space filled with / fluid-filled pleural space; bare layering /
# subpulmonic / basilar fluid; degree of effusion; suspect…possibly effusion;
# not entirely clear … due to PE; findings-level and/or effusion.
# FP trim: excluding; removal of; header ": No"; contralateral "none";
# "not well seen"; "? recurrent" indication; non-pleural layering fluid.
# Negation: clause-local tempered scan from .!;:\n / although|however|but.
# "?" is a starter only when not "? recurrent" (indication r/o).
# ==============================================================================

(?:
    (?:^|[.!;:\n]|\b(?:although|however|but)\s+|\?(?!\s*recurrent\b))
    (?:
        # persistence exceptions: "no … change|increase|decrease", "not resolved",

        (?:

        )
        |
        (?:
            (?=\bnot\s+entirely\s+clear)

        (?:

        (?:
            (?=\bconcerning\s+for\s+(?:pneumonia|consolidat|atelecta|opacit|infiltrat|infect))
            [^.!?;:\n]

        |
        (?:
            (?!\b(?:

    (?<!\?)(?<!\?\s)(?<!history:\s)(?<!hx:\s)(?<!insertion\sfor\s)(?<!tube\sfor\s)

        |
