# Thoracic aortic wall / knob calcification (not valve, not bare athero, not abdominal-only).
# Prompt: thoracic aorta / arch / knob / knuckle; atherosclerotic aorta OK; exclude valves,
# coronaries, non-aortic vessels, abdominal-only aorta, history-only inference.
# FN-fit: also allow "thoracic and (upper) abdominal aorta/aortic" — lookbehind (?<!abdominal\s)
# alone blocked those spans even when thoracic was explicit.
# FN-fit (fn_mine_fit_shared): long-span for aortic arch/knob/wall + forward bare aorta;
# athero/vascular → bare aorta reverse; short-span 6→10; porcelain aorta;
# guards: non/not/valve before calcif; granuloma/bronchial/hilar after calcif.
# FN-fit pass-2 hunt TPs: aortoscler*; pipe-stem aorta; arch-first (tight);
# PATH + calcfication* / clacified / \bcalc\b / Ca++.

    # Shared anatomy atoms (referenced via duplication; VERBOSE has no defines).
    # TAB = thoracic/ascending/descending and (upper) abdominal aorta/aortic.
    # PATH = calcific / atherosclerotic / atheroma / sclero / plaque
    #      + calcfication* (missing-i) / clacified / \bcalc\b / Ca++.

    # ==============================================================================
    # 0. NAMED SYNONYMS (circumferential / sclerotic / pipe-stem aorta)
    # ==============================================================================
    \bporcelain\s+aorta\b

    |

    \bpipe[-\s]?stem\s+aorta\b

    |

    (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baortoscler\w*\b

    |

    # ==============================================================================
    # 1. LONG-SPAN: named thoracic / aortic arch-knob-wall / bare aorta -> PATH
    #    Recovers distant measuring / tortuosity / unfolding / as-well-as clauses.
    # ==============================================================================
    (?:
        \b(?:ascending|descending|thoracic|thoracoabdominal|thoraco-abdominal)\s+aort(?:a|ic)\b
        | \b(?:ascending|descending|thoracic)\s+and\s+(?:upper\s+)?abdominal\s+aort(?:a|ic)\b
        | \baortic\s+(?:arch|knob|knuckle|root|wall)\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baorta\b
    )
    (?!\s*(?:valve|valvular))
    (?:\W+\w+){0,14}\W+
    (?<!non\s)(?<!non-)(?<!not\s)(?<!valve\s)
    (?:
        \b(?:calcif\w*|calcfication\w*|calcinos\w*|calcium|clacified|calc|atherosclero\w*|atheroma\w*|sclero\w*|plaque\w*)\b
        | \bca\+\+
    )
    (?!\s*(?:valve|valvular|pleural|lymph|granuloma\w*|bronchial|hilar))

    |

    # ==============================================================================
    # 2. LONG-SPAN reverse: PATH -> named thoracic / aortic arch-knob-wall
    #    (bare aorta only via §2b athero/vascular — avoids cyst/granuloma FPs)
    # ==============================================================================
    (?<!non\s)(?<!non-)(?<!not\s)(?<!valve\s)
    (?:
        \b(?:calcif\w*|calcfication\w*|calcinos\w*|calcium|clacified|calc|atherosclero\w*|atheroma\w*|sclero\w*|plaque\w*)\b
        | \bca\+\+
    )
    (?!\s*(?:valve|valvular|pleural|lymph|granuloma\w*|bronchial|hilar))
    (?:\W+\w+){0,14}\W+
    (?<!\bno\s)
    (?:
        \b(?:ascending|descending|thoracic|thoracoabdominal|thoraco-abdominal)\s+aort(?:a|ic)\b
        | \b(?:ascending|descending|thoracic)\s+and\s+(?:upper\s+)?abdominal\s+aort(?:a|ic)\b
        | \baortic\s+(?:arch|knob|knuckle|root|wall)\b
    )
    (?!\s*(?:valve|valvular))

    |

    # ==============================================================================
    # 2b. Athero / vascular calc -> bare aorta (hunt: vascular…aorta; atheromatous…aorta;
    #     calcified atherosclerotic changes … aorta)
    # ==============================================================================
    (?:
        \bvascular\s+calcification\w*\b
        | \b(?:calcified\s+)?(?:atherosclero\w*|atheroma\w*)\b
    )
    (?:\W+\w+){0,12}\W+
    (?<!\bno\s)
    (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baorta\b
    (?!\s*(?:valve|valvular))

    |

    # ==============================================================================
    # 3. SHORT-SPAN: anatomy -> pathology (≤10 intervening words; was 6)
    #    aorta/aortic (not abdominal; not valve); knob/knuckle; OCR "aortais".
    #    Deliberately NOT aort\w* (blocks aortico / aortopulmonary window LN).
    # ==============================================================================
    (?:
        \baortic\s+(?:arch|knob|knuckle|root|wall)\b
        | \b(?:thoracoabdominal|thoraco-abdominal)\s+aort(?:a|ic)\b
        | \b(?:ascending|descending|thoracic)\s+and\s+(?:upper\s+)?abdominal\s+aort(?:a|ic)\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baorta\s*is\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baortais\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baorta\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baortic\b
        | \b(?:knob|knuckle)\b
    )
    (?!\s*(?:valve|valvular))
    (?:\W+\w+){0,10}\W+
    (?<!non\s)(?<!non-)(?<!not\s)(?<!valve\s)
    (?:
        \b(?:calcif\w*|calcfication\w*|calcinos\w*|calcium|clacified|calc|atherosclero\w*|atheroma\w*|sclero\w*|plaque\w*)\b
        | \bca\+\+
    )
    (?!\s*(?:valve|valvular|pleural|lymph|granuloma\w*|bronchial|hilar))

    |

    # ==============================================================================
    # 3b. Arch-first without "aortic" prefix (tight gap; zygomatic/neural/costal blocked)
    #     Hunt: "Arch calcifications are again noted."
    # ==============================================================================
    (?<!zygomatic\s)(?<!neural\s)(?<!costal\s)
    \barch\b
    (?!\s*(?:valve|valvular))
    (?:\W+\w+){0,1}\W+
    (?<!non\s)(?<!non-)(?<!not\s)(?<!valve\s)
    (?:
        \b(?:calcif\w*|calcfication\w*|calcinos\w*|calcium|clacified|calc|atherosclero\w*|atheroma\w*|sclero\w*|plaque\w*)\b
        | \bca\+\+
    )
    (?!\s*(?:valve|valvular|pleural|lymph|granuloma\w*|bronchial|hilar))

    |

    # ==============================================================================
    # 4. SHORT-SPAN: pathology -> anatomy (keep ≤6; long reverse / §2b cover distant)
    # ==============================================================================
    (?<!non\s)(?<!non-)(?<!not\s)(?<!valve\s)
    (?:
        \b(?:calcif\w*|calcfication\w*|calcinos\w*|calcium|clacified|calc|atherosclero\w*|atheroma\w*|sclero\w*|plaque\w*)\b
        | \bca\+\+
    )
    (?!\s*(?:valve|valvular|pleural|lymph|granuloma\w*|bronchial|hilar))
    (?:\W+\w+){0,6}\W+
    (?<!\bno\s)
    (?:
        \baortic\s+(?:arch|knob|knuckle|root|wall)\b
        | \b(?:thoracoabdominal|thoraco-abdominal)\s+aort(?:a|ic)\b
        | \b(?:ascending|descending|thoracic)\s+and\s+(?:upper\s+)?abdominal\s+aort(?:a|ic)\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baorta\b
        | (?<!abdominal\s)(?<!abd\.\s)(?<!abd\s)\baortic\b
        | \b(?:knob|knuckle)\b
        | \barch\b
    )
    (?!\s*(?:valve|valvular))

    # NOTE: No bare atherosclero\w*|atheroma\w* standalone (was the dominant FP driver).
