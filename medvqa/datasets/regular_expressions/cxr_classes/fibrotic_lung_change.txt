# ==============================================================================
# fibrotic_lung_change
# Chronic fibrotic parenchymal remodeling beyond isolated focal scarring.
#
# Match: pulmonary / pulmonary interstitial / pulmonary parenchymal fibrosis;
# fibrotic-fibrosing change (interstitial/appearance/appearing/native/laterality
# +lung/distortion/disease/ILD/volume loss); honeycombing / honeycombed /
# honeycomb-type; traction/tractional bronchiectasis/bronchiolectasis; UIP /
# usual interstitial pneumonia|pneumonitis / fibrotic|fibrosing NSIP; coarse
# reticulation; diffuse/extensive fibrotic scarring; fibrotic process;
# architectural distortion via hilar retraction + longstanding scarring;
# large cicatrices / cicatricial cysts; fibrocalcific process; interstitial /
# lung / chronic / massive / radiation / end-stage / underlying / basal
# fibrosis; fibrobullous change; fibrosis of/in/at anatomic lung sites (incl.
# lobe + collapse/volume loss); findings attributed as representing/reflecting/
# secondary to/background of fibrosis; appear fibrotic; current changes
# compatible/consistent with known IPF; ILD/chronic lung disease with fibrosis.
#
# Do NOT match: cystic fibrosis; indication/history-only fibrosis (IPF
# transplant hx, ?pulm fibrosis, post-radic fibrosis in INDICATION alone);
# CLINICAL HISTORY: pulmonary fibrosis without findings-section fibrosis;
# bare ILD / interstitial markings without fibrosis attribution; isolated
# apical/biapical/linear/focal scarring; fibrothorax / non-traction
# bronchiectasis alone; atelectasis-or-fibrosis hedges for focal lines.
# ==============================================================================

\b(?:
        # Idiopathic pulmonary fibrosis (incl. pulmoary typo). Block
        # "transplant for idiopathic …" and "HISTORY: idiopathic …".
        (?<!\bno\sevidence\sof\s)
        (?<!\bwithout\sevidence\sof\s)
        (?<!\bhistory\sof\s)
        (?<!\bknown\shistory\sof\s)
        (?<!\bhx\sof\s)
        (?<!\bhistory:\s)
        (?<!\bfor\s)
        idiopathic\s+(?:pulmonary|pulmoary)\s+fibrosis
    |
        # Pulmonary / pulmonary interstitial / pulmonary parenchymal fibrosis —
        # not cystic; not "… idiopathic pulmoary …" (idiopathic handled above).
        (?<!\bno\sevidence\sof\s)
        (?<!\bwithout\sevidence\sof\s)
        (?<!\bhistory\sof\s)
        (?<!\bknown\shistory\sof\s)
        (?<!\bhx\sof\s)
        (?<!\bhistory:\s)
        (?<!\bcystic\s)
        (?<!\bidiopathic\s)
        (?:pulmonary|pulmoary)(?:\s+(?:interstitial|parenchymal))?\s+fibrosis
    |
        # Anatomically / etiologically qualified fibrosis (findings language).
        # Omit bare post-radic (fires in INDICATION without findings fibrosis).
        # radiation fibrosis (not only post-radiation); underlying; end-stage.
        # basal (not only basilar) for "idiopathic pulmonary basal fibrosis".
        (?<!\bcystic\s)
        (?:
            biapical|bibasilar|basilar|basal|perihilar|subpleural|
            extensive|severe|diffuse|end-?stage|
            underlying|
            radiation|
            post-?radiation|postradiation|post-?radiotherapy
        )\s+fibrosis
    |
        # Interstitial / lung / chronic / massive fibrosis (not cystic; not
        # HISTORY: fibrosis). Qualifiers avoid DDx-list "include … fibrosis".
        # Trailing "… and interstitial fibrosis." / bare clause-end covered.
        # Lobe-qualified fibrosis; fibrosis+collapse/volume loss of lobe;
        # fibrosis in/at bases; scattered/patchy interstitial fibrosis.
        (?<!\bno\sevidence\sof\s)
        (?<!\bwithout\sevidence\sof\s)
        (?<!\bhistory\sof\s)
        (?<!\bknown\shistory\sof\s)
        (?<!\bhx\sof\s)
        (?<!\bhistory:\s)
        (?<!\bcystic\s)
        (?:
            (?:
                biapical|bibasilar|basilar|basal|perihilar|subpleural|
                chronic|severe|mild|diffuse|extensive|stable|
                scattered|patchy|
                underlying|redemonstrat\w*\s+of
            )\s+interstitial\s+fibrosis
            |
            interstitial\s+fibrosis\s+(?:in|with|are|is|at)
            |
            # Trailing "… and interstitial fibrosis." (not DDx ", interstitial fibrosis,")
            \band\s+interstitial\s+fibrosis(?=\s*[.,;)]|$)
            |
            (?:left|right|bilateral)?\s*lung\s+fibrosis
            |
            # Lobe-qualified: "left upper lobe fibrosis" / laterality-free
            # "lower lobe fibrosis" / "right upper lobe volume loss and fibrosis"
            (?:(?:left|right|bilateral)\s+)?(?:upper|middle|lower|mid)\s+lobe\s+(?:volume\s+loss\s+and\s+)?fibrosis
            |
            # Word-order: "fibrosis of the left lung" / "fibrosis in/at the subpleural…"
            # / "fibrosis and collapse of the left upper lobe"
            # / "fibrosis and volume loss … right upper lobe"
            fibrosis\s+and\s+(?:collapse|volume\s+loss)
            |
            fibrosis\s+of\s+the\s+(?:(?:left|right|bilateral)\s+)?(?:(?:upper|middle|lower|mid)\s+lobe|lung)
            |
            fibrosis\s+(?:in|at)\s+the\s+(?:subpleural|bases?(?:\s+of\s+the\s+lungs?)?|lung\s+bases?|lower\s+lobes?)
            |
            fibrosis\s+involving
            |
            chronic\s+fibrosis
            |
            (?:progressive\s+)?massive\s+fibrosis
        )
    |
        appears\s+to\s+be\s+fibrosis|
        appear(?:s|ed|ing)?\s+(?:to\s+be\s+)?fibrotic|
        attributable\s+to\s+(?:pulmonary\s+)?fibrosis|
        (?:due\s+to|perhaps\s+due\s+to|may\s+be\s+due\s+to|secondary\s+to)\s+(?:post-?radiation\s+|postradiation\s+|radiation\s+)?fibrosis|
        fibrosis\s+or\s+bronchiectasis|
        fibrosis,\s+likely\s+representing
    |
        # Findings attributed as fibrosis (not "atelectasis or fibrosis").
        # Allow severity / underlying / radiation / parenchymal before fibrosis;
        # "reflect a background of fibrosis".
        (?:
            associated\s+with|
            reflect(?:s|ing|ive\s+of)?|
            represent(?:s|ing)?|
            presumably\s+represent(?:s|ing)?|
            probably|
            consistent\s+with|
            compatible\s+with|
            suggest(?:ing|ive\s+of)|
            (?:likely\s+)?related\s+to|
            evolving
        )\s+
        (?:(?:a\s+)?background\s+of\s+)?
        (?:known\s+)?
        (?:underlying\s+)?
        (?:mild\s+|moderate\s+|severe\s+)?
        (?:chronic\s+)?
        (?:pulmonary\s+|interstitial\s+|parenchymal\s+|radiation\s+|post-?radiation\s+|postradiation\s+)?
        fibrosis
    |
        # Current CXR changes attributed to known IPF (not bare hx IPF alone).
        (?:compatible|consistent)\s+with.{0,60}(?:patient'?s\s+)?(?:known\s+)?\bipf\b
    |
        # ILD / chronic lung disease co-mentioned with fibrosis
        (?:chronic\s+)?(?:interstitial\s+)?lung\s+disease\s+(?:and|with)\s+fibrosis
    |
        # Fibrotic / fibrosing phrasing (incl. interstitial / appearance /
        # appearing / native / laterality+lung / distortion / disease / ILD /
        # volume loss; fibrosing NSIP).
        fibrotic\s+(?:
            lung|parenchymal|changes?|remodeling|change|process|scarring|
            interstitial|appearance|appearing|native|distortion|disease|
            ild\b|
            volume\s+loss|
            (?:left|right|bilateral)\s+lung
        )|
        fibrosing\s+(?:chronic\s+)?(?:interstitial\s+)?lung\s+disease|
        fibrosing\s+parenchymal|
        fibrosing\s+nsip|
        diffuse\s+fibrotic|
        fibro-?\s*bullous
    |
        # Hallmarks
        honeycomb(?:ed|ing|-?type|\s+appearance)|
        traction(?:al)?\s+bronchi(?:ectasis|olectasis)|
        (?<!\bhistory\sof\s)(?<!\bknown\shistory\sof\s)(?<!\bhx\sof\s)(?<!\bhistory:\s)\buip\b|
        usual\s+interstitial\s+pneumoni(?:a|tis)|
        late\s+fibrotic\s+stage|
        fibrotic\s+nsip|
        coarse.{0,20}reticulation
    |
        # Extensive / remodeling scarring (not isolated apical scar / fibrothorax)
        fibrocalcific\s+process|
        chronic\s+biapical\s+fibrocalcific|
        large\s+cicatrices|
        (?:large\s+)?(?:biapical\s+)?cicatricial\s+cysts|
        extensive\s+parenchymal\s+scarring|
        (?:widespread|extensive|diffuse)\s+(?:pulmonary\s+)?scarring|
        hilar\s+retraction.{0,80}longstanding\s+scarring|
        longstanding\s+scarring.{0,80}hilar\s+retraction|
        parenchymal\s+retraction
    )
