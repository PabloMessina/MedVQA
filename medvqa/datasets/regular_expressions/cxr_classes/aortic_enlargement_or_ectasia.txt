\b(?:
        # Aorta → size/caliber (e.g. "thoracic aorta is mildly widened",
        # "ascending thoracic aorta is tortuous or mildly dilated",
        # "aorta is tortuous and generally large", "aortic arch is slightly dilated",
        # "the aorta is ectatic", "tortuous aorta and probably also vascular ectasia").
        # Clause-local gaps (no .?!;:). Use aorta/aortic word forms only (not aort\w*
        # which hits aorticopulmonary). Omit tortu*/unfold*/elongat*/prominent and bare
        # aneurysm* (sister class / prompt / history FPs). Prefer "enlarged" not
        # "enlargement" (avoids "aorta and mild enlargement" of the heart).
        # ecta(?:tic|sia|sis): ectas\w* misses ectatic; ectat\w* misses ectasia.
        # Gap 0–10 (FN-fit): calcified/tortuous/brachiocephalic padding before
        # appears|could be|mildly ectatic. "generally large" folds the adverb into
        # the size token so "aorta is very tortuous and may be generally large" hits.
        # Ban stent/graft/mediastinum bridges (widened mediastinum after aortic stent).
        # (?:if\s+not\s+)?ecta*: understatement "tortuous, if not ectatic".
        # FN-fit: prominent in caliber (not mere prominence); typos dilitation /
        # ectsatic / extasia. Anatomy-first \bwide\b is a dedicated arm (not here:
        # "aortic window is wide" is AP-window, not enlargement).
        (?<!\bno\s)(?<!\bwithout\s)
        (?:(?:ascending|descending|thoracic|lower\s+thoracic|upper\s+thoracic|visualized(?:\s+thoracic)?|thoracoabdominal|thoraco-abdominal)\s+)?(?:(?<=thoracoabdominal\s)|(?<=thoraco-abdominal\s)|(?<!abdominal\s))(?:aorta\b|aortic\b)
        (?:[^\w.!?;:]+(?!(?:without|no|not|denies|absent|stent|graft|mediastin\w*|cardiac|silhouette|heart)\b)\w+){0,10}[^\w.!?;:]+
        (?<!\bwithout\s)(?<!\bno\s)(?<!\bnot\s)
        (?:enlarged|(?:dilat\w*|dilitation)|(?:if\s+not\s+)?(?:ecta(?:tic|sia|sis)|ectsatic|extasia)|(?<!non-)aneurysmal|widen(?:ed|ing)|(?:generally\s+)?\blarge\b|increased(?:\s+in)?\s+(?:caliber|diameter|size)|prominent(?:ly)?\s+in\s+(?:caliber|diameter|size))\b
        (?!\s+on\s+(?:the\s+)?(?:prior\s+|chest\s+|recent\s+)?(?:CT|CTA|MRI|MRA)\b)

        |

        # Aortic contour → aneurysmal dilatation (longer clause-local gap).
        (?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)
        aortic\s+contour\b
        (?:[^\w.!?;:]+(?!(?:without|no|not|denies|absent)\b)\w+){0,10}[^\w.!?;:]+
        aneurysm(?:al)?\b

        |

        # Size → aorta with short clause-local modifier gap (e.g. "mildly dilated thoracic
        # aorta", "moderately widened and elongated thoracic aorta", "ectatic unfolded
        # thoracic aorta", "Torturous/ectatic thoracic aorta"). No bare enlarg* with a
        # long gap (cardiac FP). No large/wide with a long gap ("large opacity ... aortic").
        # (?<!/) blocks "process/dilated aorta" but not slash-joined ectatic/ectasia.
        # Ban mediastinal-level / cardiac-silhouette bridges (widening at the level of
        # the aortic arch; enlarged silhouette ... aorta).
        (?<!\bno\s)(?<!\bno\ssignificant\s)(?<!\bno\sdefinite\s)(?<!\bno\saneurysmal\s)(?<!\bno\ssignificant\saneurysmal\s)(?<!\bwithout\s)(?<!\bnot\s)
        (?:
            (?<!/)(?:dilat\w*|dilitation|aneurysmal|widen(?:ed|ing)|increased(?:\s+in)?\s+(?:caliber|diameter|size))
            |
            (?:ecta(?:tic|sia|sis)|ectsatic|extasia)
        )

        # "ectatic and mildly tortuous ascending and descending …" covered by tight-adj arm.
        (?:[^\w.!?;:]+(?!without\b|no\b|not\b|denies\b|absent\b|cardiac\b|heart\b|atrium\b|ventricle\b|reflect\b|suggest\b|secondary\b|by\b|opacity\b|hernia\b|hiatal\b|effusion\b|level\b|mediastin\w*|silhouette\b|extent\b|definition\b|ill\b)\w+){0,5}[^\w.!?;:]+
        (?:(?:ascending|descending|thoracic|lower\s+thoracic|upper\s+thoracic|visualized(?:\s+thoracic)?|thoracoabdominal|thoraco-abdominal)\s+)?(?:(?<=thoracoabdominal\s)|(?<=thoraco-abdominal\s)|(?<!abdominal\s)(?<!infrarenal\s))(?:aorta\b|aortic\b)
        (?!\s+(?:versus|vs\.?\b))

        |

        # Enlargement / dilatation / ectasia / widening / aneurysmal dilatation /
        # increased caliber OF aorta. Allow co-mentioned tortuosity
        # ("enlargement and tortuosity of the ascending thoracic aorta",
        # "ectasia/tortuosity of the thoracic aorta"). Ban with/cardiac/athero/
        # mediastinal-level bridges so "enlarged with mild tortuosity of the aorta"
        # and "enlarged and atherosclerotic changes of the aorta" / "widening at
        # the level of the aortic arch" fail. "No significant aneurysmal dilatation".
        (?<!\bno\s)(?<!\bno\ssignificant\s)(?<!\bno\sdefinite\s)(?<!\bno\sevidence\sof\s)(?<!\bno\saneurysmal\s)(?<!\bno\ssignificant\saneurysmal\s)(?<!\bwithout\s)(?<!\bwithout\ssignificant\s)(?<!\bnot\s)
        (?:enlarg\w*|dilat\w*|dilitation|(?:ecta(?:tic|sia|sis)|ectsatic|extasia)|widening|aneurysm(?:al)?\s+(?:dilat\w*|enlarg\w*)|increase(?:d)?(?:\s+in)?(?:\s+the)?\s+(?:caliber|diameter|size))
        (?:[^\w.!?;:]+(?!without\b|no\b|not\b|denies\b|absent\b|with\b|cardiac\b|heart\b|silhouette\b|atherosclero\w*|calcifi\w*|mediastin\w*|level\b)\w+){0,3}[^\w.!?;:]+of[^\w.!?;:]+
        (?:
            # thoracic/ascending already named, then optional "and (upper) abdominal aorta"
            (?:the[^\w.!?;:]+)?(?:(?:ascending|descending|thoracic|thoracoabdominal|thoraco-abdominal)[^\w.!?;:]+)+(?:and(?:/or)?[^\w.!?;:]+(?:the[^\w.!?;:]+)?(?:(?:upper|lower|abdominal)[^\w.!?;:]+)*)?aorta\b
            |
            (?:the[^\w.!?;:]+)?
            (?:(?:ascending|descending|thoracic|visualized|lower|upper|thoracoabdominal|thoraco-abdominal)[^\w.!?;:]+)*
            (?:(?<=thoracoabdominal\s)|(?<=thoraco-abdominal\s)|(?<!abdominal\s)(?<!infrarenal\s))(?:aorta\b|aortic\b)
        )

        |

        # FN-fit: ectasia, tortuosity, and athero/calcifications of the aorta.
        # Dedicated so OF-arm keeps calcifi/athero bans ("enlarged and atherosclerotic
        # changes of the aorta").
        (?<!\bno\s)(?<!\bno\ssignificant\s)(?<!\bwithout\s)(?<!\bnot\s)
        (?:ecta(?:tic|sia|sis)|ectsatic|extasia)
        (?:[^\w.!?;:]+(?!without\b|no\b|not\b|denies\b|absent\b|cardiac\b|heart\b|mediastin\w*|level\b|with\b)\w+){0,8}[^\w.!?;:]+of[^\w.!?;:]+
        (?:
            (?:the[^\w.!?;:]+)?(?:(?:ascending|descending|thoracic|thoracoabdominal|thoraco-abdominal)[^\w.!?;:]+)+(?:and(?:/or)?[^\w.!?;:]+(?:the[^\w.!?;:]+)?(?:(?:upper|lower|abdominal)[^\w.!?;:]+)*)?aorta\b
            |
            (?:the[^\w.!?;:]+)?
            (?:(?:ascending|descending|thoracic|visualized|lower|upper|thoracoabdominal|thoraco-abdominal)[^\w.!?;:]+)*
            (?:(?<=thoracoabdominal\s)|(?<=thoraco-abdominal\s)|(?<!abdominal\s)(?<!infrarenal\s))(?:aorta\b|aortic\b)
        )

        |

        # FN-fit: "Enlargement at the aortic arch" (OF-arm is of-only; not "at the level").
        # Ban cardiac/heart/silhouette prefixes (LV/cardiac enlargement near the arch).
        (?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)
        (?<!\bcardiac\s)(?<!\bheart\s)(?<!\bsilhouette\s)(?<!\bventricular\s)(?<!\batrial\s)(?<!\bmediastinal\s)
        enlarg\w+\s+at\s+(?:the\s+)?(?:(?:ascending|descending|thoracic)\s+)?aortic\s+(?:arch|knob)\b

        |

        # FN-fit: anatomy-first wide (knob/arch/aorta), not aortic window/hiatus/valve.
        (?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)
        (?:aorta\b|aortic\s+(?:knob|arch|contour)\b)
        (?:[^\w.!?;:]+(?!(?:without|no|not|denies|absent|window|hiatus|valve|nipple)\b)\w+){0,4}[^\w.!?;:]+
        \bwide\b

        (?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)(?<!/)

        (?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)
        (?:(?:thoracoabdominal|thoraco-abdominal)\s+)?(?:(?<=thoracoabdominal\s)|(?<=thoraco-abdominal\s)|(?<!abdominal\s))

        (?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)

            (?:[^\w.!?;:]+(?!(?:without|no|not|denies|absent)\b)\w+){0,8}[^\w.!?;:]+

        (?<!\bno\s)(?<!\bwithout\s)

        (?:[^\w.!?;:]+(?!(?:without|no|not|denies|absent)\b)\w+){0,8}[^\w.!?;:]+
