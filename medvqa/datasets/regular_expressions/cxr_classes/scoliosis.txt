\b(?:
        # ==============================================================================
        # FN-fit (fn_mine_fit_shared): dextro/levoconvex(+ity); scolosis typo;
        #   thoracic curvature with convexity to L/R; apex-left/right angulation.
        # Pass-2 hunt: regional `curve`; optional S-shaped hyphen; right-convex;
        #   extra-l typos (scloliosis / levoscliosis); truncated scolios; config-first
        #   S-shaped; level adjective in curvature-of-spine.
        # ==============================================================================
        # 1. Explicit diagnostic terms (incl. levo-/dextro-/roto-/kypho- prefixes)
        #    sc?ol(?:io)?(sis|tic) catches "scoliosis", typo "scolosis" (missing 'i'),
        #    and "levosoliosis" (missing 'c'). Prefixed adjectives (dextroscoliotic).
        #    scl(?:ol)? catches extra-l "scloliosis" and missing-o "levoscliosis".
        #    \w* covers plurals (scolioses).
        # ==============================================================================
        (?:[a-z]*(?:sc?ol|scl(?:ol)?)(?:io)?(?:sis|tic))\w*
        |
        # ==============================================================================
        # 2. Truncated dictation: "scoli" / "scolios" as a standalone token
        #    Trailing \b keeps this from swallowing "scoliosis" (arm 1).
        # ==============================================================================
        scoli(?:os)?
        |
        # ==============================================================================
        # 3. Compound or spaced directional curvature words
        #    e.g. dextrocurvature / "dextro curvature" of the thoracic spine
        # ==============================================================================
        (?:dextro|levo)\s*curvatures?
        |
        # ==============================================================================
        # 3b. Directional compound convex adjectives/nouns (FN-fit hunt)
        #     dextroconvex / levoconvex / dextroconvexity / levoconvexity
        # ==============================================================================
        (?:dextro|levo)\s*convex(?:it(?:y|ies))?
        |
        # ==============================================================================
        # 4. Descriptive lateral / directional curvature (token-local; no cross-clause jump)
        #    Optional "convex" between direction and curvature:
        #    "rightward convex curvature centered along the mid thoracic spine"
        # ==============================================================================
        (?:lateral|leftward|rightward)\s+(?:convex\s+)?curvatures?
        |
        # ==============================================================================
        # 5. Shape descriptors — S-shaped spine / curvature / deformity
        #    Optional reverse(d) prefix; hyphen or space ("S shaped"); optional
        #    regional/level/convex adjectives between S-shaped and the noun.
        # ==============================================================================

        # 5b. Predicative S-shaped spine ("The spine is S-shaped" / "S shaped")
        # ==============================================================================
        spine\s+is\s+(?:revers(?:e|ed)\s+)?s[- ]shaped
        |
        # ==============================================================================
        # 5c. Curvature/angulation of a named spinal region in an S-shaped configuration
        #     (not bare "S-shaped configuration" — that can be hardware/device shape)
        # ==============================================================================
        (?:curvatures?|angulation)\s+of\s+(?:the\s+)?(?:(?:upper|mid(?:dle)?|lower)\s+)?(?:thoracic|lumbar|thoracolumbar)\s+spine\s+in\s+an?\s+(?:revers(?:e|ed)\s+)?s[- ]shaped\s+configuration

        # 5d. Config-first reverse of 5c: "S-shaped configuration of the thoracic spine"

        # ==============================================================================
        (?:revers(?:e|ed)\s+)?s[- ]shaped\s+configuration\s+of\s+(?:the\s+)?(?:(?:upper|mid(?:dle)?|lower)\s+)?(?:thoracic|lumbar|thoracolumbar)\s+spine

        # 6. Curvature/convexity of the (regional) spine — exclude pure kyphotic/lordotic

        # ==============================================================================

        |
        # ==============================================================================
        # 6b. Regional thoracic/spinal curvature with trailing convexity-to-side clause
        #     e.g. "mid thoracic curvature with convexity to the right"
        #     (requires spinal/thoracic region so mediastinal/heart convexity alone fails)
        # ==============================================================================

        # ==============================================================================

        # ==============================================================================

        # 10b. Word-order reverse: right/left convex (hyphen or space) thoracic curvature
        #      e.g. "right convex thoracic curvature" / "right-convex thoracic curvature"
        # ==============================================================================
        (?:right|left)[-\s]convex\s+(?:(?:upper|mid(?:dle)?|lower)\s+)?(?:thoracic|lumbar|thoracolumbar)\s+(?:spin(?:e|al)\s+)?curvatures?
        |
        # ==============================================================================
        # 11. Deviation of the spine only (not trachea / aorta / heart)
        # ==============================================================================
        deviations?\s+(?:of\s+(?:the\s+)?)?(?:(?:thoracic|lumbar|thoracolumbar)\s+)?spine
