        \bunchanged\s+appearance\s+of\s+(?:the\s+)?known\s+
        (?:left\s+|right\s+|bilateral\s+|bilat\.?\s+|anterior\s+|posterior\s+|
           lateral\s+|sided\s+|left-sided\s+|right-sided\s+|multiple\s+)*
        ribs?\s+fractur\w*
        |

        # 6c2. Known/previous rib fractures affirmed on the current exam
        # (match starts at known/previous so history lookbehinds do not fire;
        #  history-only "Known rib fractures." lacks the affirmation and misses)
        \b(?:known|previous)\s+
        (?:
            (?:left|right|bilateral|bilat\.?|anterior|posterior|lateral|
               sided|left-sided|right-sided|multiple|and|
               scapular|clavicular|sternal|scapul\w*)
            \s+
        ){0,8}
        \bribs?\b
        (?:
            \s+(?:and|scapular|clavicular|sternal|scapul\w*|left|right)
        ){0,6}
        \s+fractur\w*
        \s+(?:are|is|were|was|remain(?:s|ed)?)\s+
        (?:re-?demonstrated|again\s+(?:demonstrated|seen|noted|visualized|identified)|
           seen|noted|visible|unchanged|stable|demonstrated)
        |

        # 6d. Rib … in the process of fracturing (pathologic)
        \b(?:ribs?|costal)\b
        (?:
            [\s,/\-():]+
            (?!(?:no|not|without)\b)
            \w+
        ){0,16}
        [\s,/\-():]+
        (?:in\s+the\s+)?process\s+of\s+fracturing\b
        |

        # 6e. Deformities involving … rib … due to (old) fractures
        \bdeformit(?:y|ies|es)\b
        (?:
            [\s,/\-():]+
            (?!(?:no|not|without|resection)\b)
            \w+
        ){0,20}
        [\s,/\-():]+
        \b(?:ribs?|costal)\b
        (?:
            [\s,/\-():]+\w+
        ){0,16}
        [\s,/\-():]+
        (?:probably\s+|possibly\s+|likely\s+)?(?:due\s+to|from|representing)\s+
        (?:old\s+|prior\s+|remote\s+|healed\s+)?fractur\w*
        |

        # 6f. Rib(s) demonstrate / show callus formation … (healing) fractures
        \b(?:ribs?|costal)\b
        (?:
            [\s,/\-():]+
            (?!(?:no|not|without)\b)
            \w+
        ){0,8}
        [\s,/\-():]+
        (?:demonstrate|demonstrates|show|shows|showing|with)\s+
        callus(?:es)?(?:\s+formation)?
        (?:
            [\s,/\-():]+\w+
        ){0,6}
        [\s,/\-():]+
        (?:suggesting|suggestive\s+of|compatible\s+with|consistent\s+with|
           representing)?
        [\s,/\-():]*
        (?:healing\s+|healed\s+|old\s+|prior\s+)?fractur\w*
        |

        # 6g. Callus associated with / projecting over … rib (sequela, no fx token)
        \bcallus(?:es)?\b
        (?:
            [\s,/\-():]+
            (?:associated|relating|related|projecting|seen|noted|identified|
               formation|stable|is|are|was|were|with|over|about|around|along|to|the|
               a|an|left|right|posterior|anterior|lateral|medial|aspect|of|and|
               several|lower|upper|sided|left-sided|right-sided|
               \d+(?:st|nd|rd|th)|\d+)
        ){1,12}
        [\s,/\-():]+
        \b(?:ribs?|costal)\b
        |

        # 7. Rib discontinuities (chronic / displaced fracture proxy)
        \b(?:ribs?)\s+discontinuit(?:y|ies)\b
        |

        # 8. Flail chest
        \bflail\s+chest\b
    )
