# Residual mid-node calcification (umbrella).
# Prompt: explicit calcified / calcific / calcification / calcinosis wording
# (incl. unlocalized, costochondral, lymph-node, soft-tissue). Do NOT infer from
# bare athero/sclerotic change, nodules/plaques/adenopathy, normal bone,
# osteopenia, or metal devices. Children cover aorta / calcified nodule /
# cardiac-valvular / pleural — keep this leaf for residual + broad calcific root.
# Precision: fixed-width negation / prior-exam lookbehinds (Python re).

# ==============================================================================
# 1. Mid-word / residual forms children miss
# ==============================================================================
\b(?:
        fibrocalcif\w*
        | chondrocalcinosis
        | calcinos\w*
        # OCR / typo of calcification
        | calicif\w*
        # explicit mineral deposition (prompt-adjacent calcific wording)
        | hydroxyapatite
        # soft-tissue / endobronchial calcific deposits (phleboliths; broncholiths; ossific densities)
        | phlebolith\w*
        | broncholith\w*
        | ossific\s+densit\w*
)\b
# pinna chondrocalcinosis is head CT, not current CXR
(?!\s+of\s+the\s+(?:left|right)\s+pinna)

|

# ==============================================================================
# 2. Costochondral / costal-cartilage mineralization or ossification
# Prompt residual (not bare bone mineraliz*; not DISH / ALL / HO / fracture fragment)
# ==============================================================================
\bmineraliz\w*\s+(?:of\s+)?(?:the\s+)?
(?:(?:left|right|anterior|posterior|first|second|third|fourth|fifth|1st|2nd|3rd|asymmetric)\s+)*
(?:costochondral\b|costal\s+cartilag\w*|chondral\s+cartilag\w*)
|
\bcostochondral\b(?:\s+\w+){0,4}\s+mineraliz\w*\b
|
\b(?:costal|chondral)\s+cartilag\w*(?:\s+\w+){0,3}\s+mineraliz\w*\b
|
# "ossification of the (anterior left) costochondral/costal cartilage"
(?<!\bno\s)(?<!\bwithout\s)(?<!\bnot\s)
\bossification\s+of\s+(?:the\s+)?
(?:(?:left|right|anterior|posterior|first|second|third|fourth|fifth|1st|2nd|3rd|asymmetric)\s+)*
(?:costochondral|costal|chondral)\s+cartilag\w*
|
# "costal cartilage ossification" / "costochondral ossification"
\b(?:costochondral|costal|chondral)\s+cartilag\w*(?:\s+\w+){0,3}\s+ossification\b
|
\bcostochondral\s+ossification\b

|
