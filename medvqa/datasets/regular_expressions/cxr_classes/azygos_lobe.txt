
    # 1. Standard phrasing (with optional specific intervening words)
    # Matches: "azygos lobe", "azygous fissure", "azygos accessory lobe",
    # "azigos vein fissure", "azygos pseudofissure", "azygos pseudo-fissure"
    \b(?:azyg|azig)\w*(?:\s+(?:vein|accessory))?\s+(?:pseudo[\s-]*)?(?:lobe|fissure)

    # Matches: "lobe of the azygos", "fissure of the azygos",
    # "pseudofissure of the azygos"
    \b(?:pseudo[\s-]*)?(?:lobe|fissure)\s+of\s+(?:the\s+)?(?:azyg|azig)\w*
